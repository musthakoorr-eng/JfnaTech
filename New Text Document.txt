        async function initFirebase() {
            try {
                if (typeof __firebase_config === 'undefined') return;
                const firebaseConfig = JSON.parse(__firebase_config);
                const app = initializeApp(firebaseConfig);
                auth = getAuth(app);
                db = getFirestore(app);

                // Rule 3: Auth Before Queries
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }

                onAuthStateChanged(auth, (user) => {
                    currentUser = user;
                    if (user) {
                        if (user.email) {
                            activeGmail = user.email;
                            updateGmailDisplays();
                        }
                        subscribeToPersistentData(user.uid);
                    }
                });
            } catch (err) {
                console.error("Firebase init error:", err);
            }
        }

        function updateGmailDisplays() {

        // Firestore Real-time Sync (Rule 1 & Rule 2 Compliant)
        function subscribeToPersistentData(uid) {
            if (!db || !uid) return;

            // 1. User Profile & Saved Courses Document
            const profileDocRef = doc(db, 'artifacts', appId, 'users', uid, 'portalProfile', 'main');
            onSnapshot(profileDocRef, (snap) => {
                if (snap.exists()) {
                    const data = snap.data();
                    userProfileData = {
                        displayName: data.displayName || 'Musthakoor',
                        department: data.department || 'Student / Applicant - UoJ',
                        savedCourses: data.savedCourses || []
                    };
                    if (data.gmail) {
                        activeGmail = data.gmail;
                        updateGmailDisplays();
                    }
                    document.getElementById('profileDisplayName').value = userProfileData.displayName;
                    document.getElementById('profileDepartment').value = userProfileData.department;
                    document.getElementById('contactName').value = userProfileData.displayName;
                    document.getElementById('applyApplicantName').value = userProfileData.displayName;
                } else {
                    // Initialize default record for musthakoorr@gmail.com
                    setDoc(profileDocRef, {
                        gmail: activeGmail,
                        displayName: userProfileData.displayName,
                        department: userProfileData.department,
                        savedCourses: []
                    }, { merge: true }).catch(console.error);
                }
                renderCourses();
                renderSavedCoursesModal();
            }, (err) => console.error("Profile snapshot error:", err));

            // 2. User Submissions (Inquiries & Job Applications)
            const submissionsColRef = collection(db, 'artifacts', appId, 'users', uid, 'submissions');
            onSnapshot(submissionsColRef, (snapshot) => {
                const items = [];
                snapshot.forEach(docSnap => {
                    items.push({ id: docSnap.id, ...docSnap.data() });
                });
                // Sort in JS memory (Rule 2)
                items.sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
                userHistoryList = items;
                renderHistoryModal();
            }, (err) => console.error("Submissions snapshot error:", err));
        }

        // Render Functions
        function renderCourses() {
            const grid = document.getElementById('coursesGrid');
            const query = document.getElementById('courseSearchInput').value.toLowerCase();

            const filtered = coursesData.filter(c => {
                const matchesFac = selectedCourseFaculty === 'All' || c.faculty === selectedCourseFaculty;
                const matchesSearch = c.title.toLowerCase().includes(query) || c.desc.toLowerCase().includes(query);
                return matchesFac && matchesSearch;
            });

            grid.innerHTML = filtered.map(course => {
                const isSaved = userProfileData.savedCourses.includes(course.id);
                return `
                    <div class="bg-white rounded-3xl p-6 shadow-sm border border-slate-200 flex flex-col justify-between hover:shadow-md transition-all">
                        <div>
                            <div class="flex justify-between items-start gap-2 mb-3">
                                <span class="text-xs font-bold bg-amber-50 text-amber-700 px-3 py-1 rounded-full">${course.faculty}</span>
                                <span class="text-xs font-medium text-slate-500 flex items-center">
                                    <i data-lucide="clock" class="w-3.5 h-3.5 mr-1"></i>${course.duration}
                                </span>
                            </div>
                            <h3 class="font-bold text-lg text-uojNavy mb-2">${course.title}</h3>
                            <p class="text-sm text-slate-600 leading-relaxed mb-5">${course.desc}</p>
                        </div>
                        <button onclick="toggleSaveCourse('${course.id}')" class="w-full py-2.5 px-4 rounded-xl text-xs font-bold flex items-center justify-center space-x-2 transition-all ${isSaved ? 'bg-emerald-50 text-emerald-700 border border-emerald-200' : 'bg-slate-100 hover:bg-uojNavy hover:text-white text-slate-700'}">
                            <i data-lucide="${isSaved ? 'bookmark-check' : 'bookmark'}" class="w-4 h-4"></i>
                            <span>${isSaved ? 'சேமிக்கப்பட்டது (Saved)' : 'பாடநெறியைச் சேமி (Bookmark)'}</span>

        function renderSavedCoursesModal() {
            const container = document.getElementById('savedCoursesContainer');
            if (!userProfileData.savedCourses.length) {

        // SPA Page Router
        window.navigateTo = function(pageId) {
            document.querySelectorAll('.page-section').forEach(sec => sec.classList.add('hidden'));
            const target = document.getElementById(`page-${pageId}`);
            if (target) target.classList.remove('hidden');

            // Update Desktop Nav Active State
            document.querySelectorAll('.nav-link').forEach(btn => {
                btn.classList.remove('nav-active');
                if (btn.getAttribute('data-nav') === pageId) {
                    btn.classList.add('nav-active');
                }
            });

            // Update Mobile Nav Active State
            document.querySelectorAll('.mobile-nav-link').forEach(btn => {
                btn.classList.remove('bg-amber-50', 'text-amber-700');
                btn.classList.add('text-slate-700');
                if (btn.getAttribute('data-nav') === pageId) {
                    btn.classList.add('bg-amber-50', 'text-amber-700');
                    btn.classList.remove('text-slate-700');
                }
            });

            document.getElementById('mobileMenu').classList.add('hidden');
            window.scrollTo({ top: 0, behavior: 'smooth' });
        };
